## Overview

This guide will help you create your first NSM application - a simple collaborative counter that multiple users can increment together in real-time.

<Steps>
  <Step title="Install NSM Framework">
    Add NSM Framework to your project using your preferred package manager:

    <CodeGroup>
      ```bash npm
      npm install @nsm/client @nsm/core
      ```

      ```bash yarn
      yarn add @nsm/client @nsm/core
      ```

      ```bash pnpm
      pnpm add @nsm/client @nsm/core
      ```

      ```bash bun
      bun add @nsm/client @nsm/core
      ```
    </CodeGroup>

    <Note>
      You'll also need to install XState v5 for state machine functionality:
      ```bash
      npm install xstate@^5.0.0
      ```
    </Note>
  </Step>

  <Step title="Create a State Machine">
    Define your application's state machine using XState v5 with NSM integration:

    ```typescript
    // counter-machine.ts
    import { createMachine, assign } from 'xstate';

    interface CounterContext {
      count: number;
      lastUpdatedBy?: string;
      participants: string[];
    }

    type CounterEvent =
      | { type: 'INCREMENT'; userId: string }
      | { type: 'DECREMENT'; userId: string }
      | { type: 'RESET'; userId: string }
      | { type: 'USER_JOINED'; userId: string }
      | { type: 'USER_LEFT'; userId: string };

    export const counterMachine = createMachine({
      /** @xstate-layout N4IgpgJg5mDOIC5QCMCGBXAXMAYBIUA2AZBUA */
      id: 'collaborativeCounter',
      initial: 'active',
      context: {
        count: 0,
        lastUpdatedBy: undefined,
        participants: []
      } satisfies CounterContext,
      states: {
        active: {
          on: {
            INCREMENT: {
              actions: assign({
                count: ({ context }) => context.count + 1,
                lastUpdatedBy: ({ event }) => event.userId
              })
            },
            DECREMENT: {
              actions: assign({
                count: ({ context }) => Math.max(0, context.count - 1),
                lastUpdatedBy: ({ event }) => event.userId
              })
            },
            RESET: {
              actions: assign({
                count: 0,
                lastUpdatedBy: ({ event }) => event.userId
              })
            },
            USER_JOINED: {
              actions: assign({
                participants: ({ context, event }) =>
                  context.participants.includes(event.userId)
                    ? context.participants
                    : [...context.participants, event.userId]
              })
            },
            USER_LEFT: {
              actions: assign({
                participants: ({ context, event }) =>
                  context.participants.filter(id => id !== event.userId)
              })
            }
          }
        }
      }
    }, {
      // XState v5 type definitions
      types: {
        context: {} as CounterContext,
        events: {} as CounterEvent
      }
    });
    ```
  </Step>

  <Step title="Initialize NSM Client">
    Create an NSM client instance and set up the collaborative application:

    ```typescript
    // counter-app.ts
    import { NSMClient } from '@nsm/client';
    import { createActor } from 'xstate';
    import { counterMachine } from './counter-machine';

    class CounterApp {
      private client: NSMClient;
      private actor: any;
      private applicationId = 'collaborative-counter';

      constructor() {
        // Initialize NSM client with Nostr configuration
        this.client = new NSMClient({
          relayUrls: [
            'wss://relay.damus.io',
            'wss://nos.lol',
            'wss://relay.nostr.band'
          ],
          // Option 1: Use NIP-07 browser extension for signing
          useNip07: true,

          // Option 2: Use private key (for development only!)
          // privateKey: process.env.NOSTR_PRIVATE_KEY,

          autoConnect: true
        });

        // Create XState actor for local state management
        this.actor = createActor(counterMachine);
      }

      async initialize() {
        // Connect to Nostr network
        await this.client.connect();

        // Set up state synchronization
        this.setupStateSynchronization();

        // Start the state machine
        this.actor.start();

        console.log('Counter app initialized!');
      }

      private setupStateSynchronization() {
        // Subscribe to remote state updates
        this.client.subscribeToApplication(this.applicationId, {
          onInteraction: (interaction) => {
            // Forward remote interactions to local state machine
            this.actor.send(interaction);
          },
          onStateUpdate: (stateUpdate) => {
            // Handle canonical state updates
            console.log('Received state update:', stateUpdate);
          },
          onError: (error) => {
            console.error('Subscription error:', error);
          }
        });

        // Listen for local state changes and broadcast them
        this.actor.subscribe((state) => {
          console.log('Local state updated:', state.context);
          // Update UI based on state changes
          this.updateUI(state.context);
        });
      }

      // Public methods for UI interaction
      async increment() {
        const userId = await this.client.getUserPublicKey() || 'anonymous';
        const event = { type: 'INCREMENT', userId } as const;

        // Send to local state machine
        this.actor.send(event);

        // Broadcast to network
        await this.client.publishInteraction({
          applicationId: this.applicationId,
          action: event.type,
          payload: { userId: event.userId }
        });
      }

      async decrement() {
        const userId = await this.client.getUserPublicKey() || 'anonymous';
        const event = { type: 'DECREMENT', userId } as const;

        this.actor.send(event);
        await this.client.publishInteraction({
          applicationId: this.applicationId,
          action: event.type,
          payload: { userId: event.userId }
        });
      }

      async reset() {
        const userId = await this.client.getUserPublicKey() || 'anonymous';
        const event = { type: 'RESET', userId } as const;

        this.actor.send(event);
        await this.client.publishInteraction({
          applicationId: this.applicationId,
          action: event.type,
          payload: { userId: event.userId }
        });
      }

      private updateUI(context: any) {
        // Update DOM elements with new state
        const counterEl = document.getElementById('counter-value');
        const participantsEl = document.getElementById('participants');
        const lastUpdatedEl = document.getElementById('last-updated');

        if (counterEl) counterEl.textContent = context.count.toString();
        if (participantsEl) participantsEl.textContent = context.participants.length.toString();
        if (lastUpdatedEl) lastUpdatedEl.textContent = context.lastUpdatedBy || 'none';
      }
    }

    // Export for use in HTML
    export { CounterApp };
    ```

    <Warning>
      Never hardcode private keys in production! Use environment variables, NIP-07 browser extensions, or secure key management systems.
    </Warning>
  </Step>

  <Step title="Connect and Interact">
    Initialize the application and connect user interactions:

    ```typescript
    // main.ts
    import { CounterApp } from './counter-app';

    async function startCounterApp() {
      const app = new CounterApp();

      try {
        // Initialize the application
        await app.initialize();

        // Set up UI event handlers
        setupUIHandlers(app);

        console.log('Collaborative counter is ready!');
      } catch (error) {
        console.error('Failed to start counter app:', error);
      }
    }

    function setupUIHandlers(app: CounterApp) {
      const incrementBtn = document.getElementById('increment-btn');
      const decrementBtn = document.getElementById('decrement-btn');
      const resetBtn = document.getElementById('reset-btn');

      incrementBtn?.addEventListener('click', () => {
        app.increment().catch(console.error);
      });

      decrementBtn?.addEventListener('click', () => {
        app.decrement().catch(console.error);
      });

      resetBtn?.addEventListener('click', () => {
        app.reset().catch(console.error);
      });
    }

    // Start the app when the page loads
    document.addEventListener('DOMContentLoaded', startCounterApp);
    ```
  </Step>

## What Just Happened?

Congratulations! You've just created your first NSM application. Here's what happened:

1. **State Machine**: You defined a deterministic state machine with increment, decrement, and reset actions
2. **Network Layer**: NSM automatically synchronized the state across all connected clients via Nostr
3. **Event Sourcing**: All state changes are automatically logged and can be replayed
4. **Collaboration**: Multiple users can interact with the same counter in real-time

## Troubleshooting

<AccordionGroup>
  <Accordion title="Connection Issues">
    If you're having trouble connecting to Nostr relays:
    - Verify your relay URLs are correct and accessible
    - Check your internet connection
    - Try different relay servers
    - Ensure your private key is valid
  </Accordion>

  <Accordion title="State Not Syncing">
    If state changes aren't appearing across clients:
    - Verify all clients are using the same `roomId`
    - Check that events are being sent correctly
    - Ensure your state machine is deterministic
    - Look for network connectivity issues
  </Accordion>

  <Accordion title="Performance Issues">
    For performance optimization:
    - Limit the frequency of state updates
    - Use debouncing for rapid user interactions
    - Consider state machine optimization
    - Monitor network usage and relay performance
  </Accordion>
</AccordionGroup>
