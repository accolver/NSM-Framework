# Real-time Chat Application

Learn how to build a sophisticated real-time chat application using NSM Framework. This example demonstrates multi-room chat, user presence, message history, and advanced state management patterns.

## Features

- 🏠 **Multi-room Support**: Create and join different chat channels
- 💬 **Real-time Messaging**: Instant message delivery across all connected users
- 👥 **User Presence**: See who's online in each channel
- 📜 **Message History**: Persistent message storage and retrieval
- 🔔 **Notifications**: Desktop notifications for new messages
- ✏️ **Typing Indicators**: Show when users are typing
- 🎨 **Rich Messages**: Support for emojis, mentions, and formatting

## Quick Implementation

Here's a simplified chat application showing the core concepts:

```typescript
// Chat State Machine
import { createMachine, assign } from 'xstate';

export const chatMachine = createMachine({
  id: 'chat',
  initial: 'connecting',
  context: {
    messages: [],
    users: [],
    currentUser: null,
    currentRoom: 'general'
  },
  states: {
    connecting: {
      on: {
        CONNECT_SUCCESS: {
          target: 'connected',
          actions: 'setUser'
        }
      }
    },
    connected: {
      on: {
        SEND_MESSAGE: {
          actions: 'addMessage'
        },
        RECEIVE_MESSAGE: {
          actions: 'receiveMessage'
        },
        USER_JOINED: {
          actions: 'addUser'
        }
      }
    }
  }
}, {
  actions: {
    setUser: assign({
      currentUser: ({ event }) => event.user
    }),
    addMessage: assign({
      messages: ({ context, event }) => [
        ...context.messages,
        {
          id: Date.now(),
          text: event.text,
          sender: context.currentUser.id,
          timestamp: Date.now()
        }
      ]
    }),
    receiveMessage: assign({
      messages: ({ context, event }) => [
        ...context.messages,
        event.message
      ]
    }),
    addUser: assign({
      users: ({ context, event }) => [
        ...context.users.filter(u => u.id !== event.user.id),
        event.user
      ]
    })
  }
});
```

```tsx
// Chat Component
import React, { useState, useEffect } from 'react';
import { NSMClient } from '@nsm/client';
import { createActor } from 'xstate';
import { chatMachine } from './chat-machine';

export const SimpleChat: React.FC = () => {
  const [client] = useState(() => new NSMClient({ useNip07: true }));
  const [actor] = useState(() => createActor(chatMachine));
  const [state, setState] = useState<any>({});
  const [message, setMessage] = useState('');

  useEffect(() => {
    const subscription = actor.subscribe(setState);

    // Initialize
    client.connect().then(() => {
      actor.send({
        type: 'CONNECT_SUCCESS',
        user: { id: 'user-1', name: 'User' }
      });
    });

    actor.start();

    return () => {
      subscription.unsubscribe();
      client.disconnect();
    };
  }, []);

  const sendMessage = async () => {
    if (!message.trim()) return;

    // Local update
    actor.send({
      type: 'SEND_MESSAGE',
      text: message
    });

    // Network broadcast
    await client.publishInteraction({
      applicationId: 'simple-chat',
      action: 'MESSAGE_SENT',
      payload: {
        text: message,
        sender: state.context?.currentUser?.id,
        timestamp: Date.now()
      }
    });

    setMessage('');
  };

  return (
    <div className="chat-app">
      <div className="messages">
        {state.context?.messages.map((msg: any) => (
          <div key={msg.id} className="message">
            <strong>{msg.sender}:</strong> {msg.text}
          </div>
        ))}
      </div>

      <div className="input-area">
        <input
          value={message}
          onChange={(e) => setMessage(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && sendMessage()}
          placeholder="Type a message..."
        />
        <button onClick={sendMessage}>Send</button>
      </div>
    </div>
  );
};
```

## Advanced Features

For a production chat application, you would extend this with:

### Multi-room Support
```typescript
const multiRoomChatMachine = createMachine({
  context: {
    rooms: new Map(),
    currentRoomId: null,
    // ... other context
  },
  states: {
    connected: {
      on: {
        CREATE_ROOM: { actions: 'createRoom' },
        JOIN_ROOM: { actions: 'joinRoom' },
        SWITCH_ROOM: { actions: 'switchRoom' }
      }
    }
  }
});
```

### User Presence
```typescript
// Track online users
const presenceService = {
  async broadcastPresence(userId: string, status: 'online' | 'away') {
    await client.publishInteraction({
      applicationId: 'chat-presence',
      action: 'PRESENCE_UPDATE',
      payload: { userId, status, timestamp: Date.now() }
    });
  }
};
```
